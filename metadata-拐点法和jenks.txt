library(ggplot2)
library(patchwork) # 用于拼图，如果没有安装请运行 install.packages("patchwork")

# ==============================================================================
# 1. 准备数据与计算
# ==============================================================================

# 提取评分
scores <- as.numeric(sc.obj$aucell)

# --- 方法 A: K-means 聚类 (k=2) ---
set.seed(123)
km_res <- kmeans(scores, centers = 2)
# 计算两个中心的中间值作为阈值
cutoff_kmeans <- mean(km_res$centers)

# --- 方法 B: 拐点法 (Knee/Inflection Point) ---
# 1. 从大到小排序
sorted_scores <- sort(scores, decreasing = TRUE)
n_points <- length(sorted_scores)
x_rank <- 1:n_points

# 2. 几何法寻找拐点 (距离对角线最远的点)
# 这是一个经典的寻找“肘/膝”点的算法 (Kneedle 算法简化版)
find_knee_point <- function(x, y) {
  # 归一化到 0-1
  x_norm <- (x - min(x)) / (max(x) - min(x))
  y_norm <- (y - min(y)) / (max(y) - min(y))
  # 计算每个点到连接首尾直线的距离
  # 在归一化坐标系中，直线方程 x - y = 0 (或 x + y = c)
  # 实际上就是找 (y_norm - x_norm) 差值最大处（或者根据曲线凹凸性找反方向）
  # 对于从大到小下降的曲线，通常是找离(0,0)到(1,1)连线最远的点
  # 这里使用简单的向量投影逻辑：
  start_point <- c(x_norm[1], y_norm[1])
  end_point <- c(x_norm[length(x)], y_norm[length(y)])
  
  # 直线向量
  line_vec <- end_point - start_point
  
  # 计算距离
  dists <- sapply(1:length(x), function(i) {
    point <- c(x_norm[i], y_norm[i])
    vec_point <- point - start_point
    # 叉乘求面积/距离
    abs(line_vec[1] * vec_point[2] - line_vec[2] * vec_point[1])
  })
  
  return(which.max(dists))
}

knee_index <- find_knee_point(x_rank, sorted_scores)
cutoff_knee <- sorted_scores[knee_index]

# ==============================================================================
# 2. 输出结果与可视化对比
# ==============================================================================

cat("\n=== 阈值计算结果 ===\n")
cat(sprintf("1. K-means 建议阈值 (分布法): %.4f\n", cutoff_kmeans))
cat(sprintf("2. 拐点法 建议阈值 (排序法):   %.4f\n", cutoff_knee))
cat("====================\n")

# --- 图 1: 直方图 (展示 K-means 切割位置) ---
p1 <- ggplot(data.frame(s=scores), aes(x=s)) +
  geom_histogram(bins=60, fill="lightgrey", color="white") +
  geom_vline(xintercept = cutoff_kmeans, color="blue", size=1, linetype="dashed") +
  annotate("text", x=cutoff_kmeans, y=Inf, label=paste0("K-means\n", round(cutoff_kmeans,3)), 
           vjust=1.5, color="blue", fontface="bold") +
  labs(title="方法一: K-means (分布)", x="AUCell Score", y="Count") +
  theme_minimal()

# --- 图 2: 排序图 (展示 拐点位置) ---
plot_data_rank <- data.frame(Rank = x_rank, Score = sorted_scores)

p2 <- ggplot(plot_data_rank, aes(x=Rank, y=Score)) +
  geom_line(color="darkgreen", size=1) +
  geom_point(x=knee_index, y=cutoff_knee, color="red", size=3) + # 标出拐点
  geom_hline(yintercept = cutoff_knee, color="red", linetype="dashed") +
  annotate("text", x=n_points*0.7, y=cutoff_knee, label=paste0("Knee Point\n", round(cutoff_knee,3)), 
           vjust=-0.5, color="red", fontface="bold") +
  labs(title="方法二: 拐点法 (排序)", x="Cell Rank", y="AUCell Score") +
  theme_minimal()

# 拼图显示
p1 + p2
