# ==============================================================================
# 步骤 1: 准备工作与计算
# ==============================================================================
library(BAMMtools) # 用于 Jenks 自然断点
library(ggplot2)   # 用于绘图

# 提取评分数据 (确保是数值型)
scores <- as.numeric(sc.obj$aucell)

# --- 方法 A: Jenks 自然断点法 (k=2) ---
# getJenksBreaks 返回 [最小值, 断点, 最大值]
jenks_breaks <- getJenksBreaks(scores, k = 2)
t_jenks <- jenks_breaks[2]

# --- 方法 B: K-means 聚类法 (k=2) ---
set.seed(123) # 设置随机种子，保证结果稳定
km_res <- kmeans(scores, centers = 2)
# 取两个聚类中心的中间值作为断点
t_kmeans <- mean(km_res$centers)

# ==============================================================================
# 步骤 2: 输出数值对比与可视化
# ==============================================================================

# 1. 在控制台打印数值
cat("\n================ 阈值计算结果 ================\n")
cat(sprintf("1. Jenks 建议阈值 (红色虚线): %.4f\n", t_jenks))
cat(sprintf("2. K-means 建议阈值 (蓝色实线): %.4f\n", t_kmeans))
cat("==============================================\n\n")

# 2. 绘制对比图
plot_data <- data.frame(Score = scores)

p <- ggplot(plot_data, aes(x = Score)) +
  # 绘制直方图
  geom_histogram(bins = 60, fill = "lightgrey", color = "white") +
  
  # 绘制 Jenks 线 (红色虚线)
  geom_vline(aes(xintercept = t_jenks, color = "Jenks"), 
             linetype = "dashed", size = 1) +
  
  # 绘制 K-means 线 (蓝色实线)
  geom_vline(aes(xintercept = t_kmeans, color = "K-means"), 
             linetype = "solid", size = 1, alpha = 0.7) +
  
  # 标注具体数值
  annotate("text", x = t_jenks, y = Inf, label = round(t_jenks, 3), 
           vjust = 2, color = "red", fontface = "bold") +
  annotate("text", x = t_kmeans, y = Inf, label = round(t_kmeans, 3), 
           vjust = 4, color = "blue", fontface = "bold") +
  
  # 设置图例和标题
  scale_color_manual(name = "算法方法", 
                     values = c("Jenks" = "red", "K-means" = "blue")) +
  labs(title = "二分类阈值对比: Jenks vs K-means",
       subtitle = "请观察哪条线更好地将双峰分开 (落在波谷)",
       x = "AUCell Score", y = "细胞数量") +
  theme_minimal()

# 显示图片
print(p)

# ==============================================================================
# 步骤 3: 决策与写入 (请根据上图结果，修改下方代码)
# ==============================================================================

# 【请选择】：查看图片后，决定使用哪个变量。
# 如果觉得 Jenks (红线) 准，就用 t_jenks
# 如果觉得 K-means (蓝线) 准，就用 t_kmeans
# 或者你自己手动指定一个数字，例如: final_cutoff <- 0.15

final_cutoff <- t_jenks  # <--- 默认设为 Jenks，你可以修改为 t_kmeans

# 执行写入 Seurat
# 大于阈值为 High (阳性)，小于等于为 Low (阴性)
sc.obj$aucell_group <- ifelse(sc.obj$aucell > final_cutoff, "High", "Low")

# 打印最终分类统计
cat("\n最终分类结果统计 (基于阈值", round(final_cutoff, 4), "):\n")
print(table(sc.obj$aucell_group))
